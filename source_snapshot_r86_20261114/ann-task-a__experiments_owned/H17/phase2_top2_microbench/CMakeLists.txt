cmake_minimum_required(VERSION 3.18)
project(H17Top2Microbench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(HWY_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(HWY_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(HWY_ENABLE_CONTRIB ON CACHE BOOL "" FORCE)
if(NOT EXISTS "${H17_HIGHWAY_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Set -DH17_HIGHWAY_SOURCE=<original cached Highway>")
endif()
add_subdirectory("${H17_HIGHWAY_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/highway-bin" EXCLUDE_FROM_ALL)
add_executable(h17_top2 h17_top2.cpp)
target_compile_options(h17_top2 PRIVATE -O3 -march=native)
target_link_libraries(h17_top2 PRIVATE hwy hwy_contrib)
